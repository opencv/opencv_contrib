set(the_description "Live visual views for OpenCV frame channels")

ocv_module_include_directories("${CMAKE_CURRENT_SOURCE_DIR}/src")

set(LIVEVIEW_HTTP_BACKEND "AUTO" CACHE STRING "HTTP backend (AUTO, CIVETWEB, BOOST, MONGOOSE)")
set_property(CACHE LIVEVIEW_HTTP_BACKEND PROPERTY STRINGS AUTO CIVETWEB BOOST MONGOOSE)
string(TOUPPER "${LIVEVIEW_HTTP_BACKEND}" LIVEVIEW_HTTP_BACKEND)

set(LIVEVIEW_VIDEO_ENCODER "AUTO" CACHE STRING "Video encoder backend (AUTO, OFF, FFMPEG)")
set_property(CACHE LIVEVIEW_VIDEO_ENCODER PROPERTY STRINGS AUTO OFF FFMPEG)
string(TOUPPER "${LIVEVIEW_VIDEO_ENCODER}" LIVEVIEW_VIDEO_ENCODER)

set(LIVEVIEW_CIVETWEB_DIR "" CACHE PATH "CivetWeb prefix")
set(LIVEVIEW_BOOST_DIR "" CACHE PATH "Boost prefix")

find_package(PkgConfig QUIET)
find_package(Threads QUIET)
ocv_check_modules(FFMPEG libavcodec libavformat libavutil libswscale QUIET)

set(CIVETWEB_FOUND FALSE)
if(PKG_CONFIG_FOUND)
    pkg_check_modules(PC_CIVETWEB QUIET IMPORTED_TARGET civetweb)
endif()
if(PC_CIVETWEB_FOUND)
    set(CIVETWEB_FOUND TRUE)
    set(CIVETWEB_INCLUDE_DIRS ${PC_CIVETWEB_INCLUDE_DIRS})
    set(CIVETWEB_LIBRARIES PkgConfig::PC_CIVETWEB)
else()
    find_path(CIVETWEB_INCLUDE_DIR NAMES civetweb.h HINTS ${LIVEVIEW_CIVETWEB_DIR} ENV CMAKE_PREFIX_PATH)
    find_library(CIVETWEB_C_LIB NAMES civetweb HINTS ${LIVEVIEW_CIVETWEB_DIR} ENV CMAKE_PREFIX_PATH)
    if(CIVETWEB_INCLUDE_DIR AND CIVETWEB_C_LIB)
        set(CIVETWEB_FOUND TRUE)
        set(CIVETWEB_INCLUDE_DIRS ${CIVETWEB_INCLUDE_DIR})
        set(CIVETWEB_LIBRARIES ${CIVETWEB_C_LIB})
    endif()
endif()

set(BOOST_FOUND FALSE)
find_path(BOOST_INCLUDE_DIR NAMES boost/beast/core.hpp HINTS ${LIVEVIEW_BOOST_DIR} ENV BOOST_ROOT)
if(BOOST_INCLUDE_DIR)
    set(BOOST_FOUND TRUE)
    set(BOOST_INCLUDE_DIRS ${BOOST_INCLUDE_DIR})
    find_library(BOOST_SYSTEM_LIB NAMES boost_system HINTS ${LIVEVIEW_BOOST_DIR} ENV BOOST_ROOT)
    set(BOOST_LIBRARIES ${BOOST_SYSTEM_LIB})
endif()

set(LIVEVIEW_SELECTED_HTTP_BACKEND "")
if(NOT LIVEVIEW_HTTP_BACKEND STREQUAL "AUTO")
    if(LIVEVIEW_HTTP_BACKEND STREQUAL "CIVETWEB" AND CIVETWEB_FOUND)
        set(LIVEVIEW_SELECTED_HTTP_BACKEND "CIVETWEB")
    elseif(LIVEVIEW_HTTP_BACKEND STREQUAL "BOOST" AND BOOST_FOUND)
        set(LIVEVIEW_SELECTED_HTTP_BACKEND "BOOST")
    elseif(LIVEVIEW_HTTP_BACKEND STREQUAL "MONGOOSE")
        set(LIVEVIEW_SELECTED_HTTP_BACKEND "MONGOOSE")
    else()
        set(_liveview_backend_warn "LiveView: requested HTTP backend '${LIVEVIEW_HTTP_BACKEND}' is not available; falling back to AUTO.")
    endif()
endif()

if(LIVEVIEW_SELECTED_HTTP_BACKEND STREQUAL "")
    if(CIVETWEB_FOUND)
        set(LIVEVIEW_SELECTED_HTTP_BACKEND "CIVETWEB")
    elseif(BOOST_FOUND)
        set(LIVEVIEW_SELECTED_HTTP_BACKEND "BOOST")
    else()
        set(LIVEVIEW_SELECTED_HTTP_BACKEND "MONGOOSE")
    endif()
endif()

set(LIVEVIEW_VIDEO_ENCODER_ENABLED FALSE)
if(LIVEVIEW_VIDEO_ENCODER STREQUAL "FFMPEG")
    if(FFMPEG_FOUND)
        set(LIVEVIEW_VIDEO_ENCODER_ENABLED TRUE)
    else()
        message(FATAL_ERROR "LiveView: LIVEVIEW_VIDEO_ENCODER=FFMPEG requires libavcodec, libavformat, libavutil, and libswscale.")
    endif()
elseif(LIVEVIEW_VIDEO_ENCODER STREQUAL "AUTO")
    if(FFMPEG_FOUND)
        set(LIVEVIEW_VIDEO_ENCODER_ENABLED TRUE)
    endif()
elseif(NOT LIVEVIEW_VIDEO_ENCODER STREQUAL "OFF")
    message(FATAL_ERROR "LiveView: unknown LIVEVIEW_VIDEO_ENCODER='${LIVEVIEW_VIDEO_ENCODER}'")
endif()

set(LIVEVIEW_HTTP_INCLUDES "")
set(LIVEVIEW_HTTP_LIBS "")
set(LIVEVIEW_HTTP_SOURCES "")

if(LIVEVIEW_SELECTED_HTTP_BACKEND STREQUAL "CIVETWEB")
    set(LIVEVIEW_HTTP_INCLUDES ${CIVETWEB_INCLUDE_DIRS})
    set(LIVEVIEW_HTTP_LIBS ${CIVETWEB_LIBRARIES})
elseif(LIVEVIEW_SELECTED_HTTP_BACKEND STREQUAL "BOOST")
    set(LIVEVIEW_HTTP_INCLUDES ${BOOST_INCLUDE_DIRS})
    set(LIVEVIEW_HTTP_LIBS ${BOOST_LIBRARIES})
    if(Threads_FOUND)
        list(APPEND LIVEVIEW_HTTP_LIBS Threads::Threads)
    endif()
elseif(LIVEVIEW_SELECTED_HTTP_BACKEND STREQUAL "MONGOOSE")
    set(MONGOOSE_VERSION "7.18")
    set(MONGOOSE_HASH "e50fe3c2d43cdd5ed96833aae88ad7b1")
    set(MONGOOSE_URL "https://github.com/cesanta/mongoose/archive/refs/tags/${MONGOOSE_VERSION}.zip")
    set(MONGOOSE_DOWNLOAD_DIR "${OpenCV_BINARY_DIR}/3rdparty/mongoose")
    set(MONGOOSE_UNPACK_DIR "${MONGOOSE_DOWNLOAD_DIR}/mongoose-${MONGOOSE_VERSION}")
    set(LIVEVIEW_HTTP_SOURCES "${MONGOOSE_UNPACK_DIR}/mongoose.c")
    set(LIVEVIEW_HTTP_INCLUDES "${MONGOOSE_UNPACK_DIR}")
    ocv_download(
            FILENAME "${MONGOOSE_VERSION}.zip"
            URL "${MONGOOSE_URL}"
            HASH "${MONGOOSE_HASH}"
            DESTINATION_DIR "${MONGOOSE_DOWNLOAD_DIR}"
            UNPACK
    )
endif()

ocv_define_module(liveview opencv_core opencv_imgcodecs opencv_imgproc WRAP python)

set(LIVEVIEW_DEFINES "HAVE_LIVEVIEW_HTTP_${LIVEVIEW_SELECTED_HTTP_BACKEND}")
if(BUILD_TESTS)
    list(APPEND LIVEVIEW_DEFINES "OCV_BUILD_TESTS")
endif()
if(LIVEVIEW_VIDEO_ENCODER_ENABLED)
    list(APPEND LIVEVIEW_DEFINES "HAVE_LIVEVIEW_VIDEO_ENCODER_FFMPEG")
endif()

if(LIVEVIEW_SELECTED_HTTP_BACKEND STREQUAL "MONGOOSE")
    target_sources(${the_module} PRIVATE ${LIVEVIEW_HTTP_SOURCES})
endif()
if(LIVEVIEW_HTTP_INCLUDES)
    list(REMOVE_DUPLICATES LIVEVIEW_HTTP_INCLUDES)
    target_include_directories(${the_module} PRIVATE ${LIVEVIEW_HTTP_INCLUDES})
endif()
if(LIVEVIEW_HTTP_LIBS)
    target_link_libraries(${the_module} PRIVATE ${LIVEVIEW_HTTP_LIBS})
endif()
if(LIVEVIEW_VIDEO_ENCODER_ENABLED)
    target_include_directories(${the_module} PRIVATE ${FFMPEG_INCLUDE_DIRS})
    target_compile_options(${the_module} PRIVATE ${FFMPEG_CFLAGS_OTHER})
    target_link_libraries(${the_module} PRIVATE ${FFMPEG_LIBRARIES})
endif()
target_compile_definitions(${the_module} PRIVATE ${LIVEVIEW_DEFINES})

if(TARGET opencv_test_liveview)
    target_sources(opencv_test_liveview PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/src/channel_store.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/frame_encoder.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/mjpeg_writer.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/route_matcher.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/video_channel_encoder.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/video_encoder.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/video_encoder_ffmpeg.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/web_backend.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/web_backend_boost.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/web_backend_civetweb.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/web_backend_mongoose.cpp"
    )
    if(LIVEVIEW_SELECTED_HTTP_BACKEND STREQUAL "MONGOOSE")
        target_sources(opencv_test_liveview PRIVATE ${LIVEVIEW_HTTP_SOURCES})
    endif()
    if(LIVEVIEW_HTTP_INCLUDES)
        target_include_directories(opencv_test_liveview PRIVATE ${LIVEVIEW_HTTP_INCLUDES})
    endif()
    if(LIVEVIEW_HTTP_LIBS)
        target_link_libraries(opencv_test_liveview PRIVATE ${LIVEVIEW_HTTP_LIBS})
    endif()
    if(LIVEVIEW_VIDEO_ENCODER_ENABLED)
        target_include_directories(opencv_test_liveview PRIVATE ${FFMPEG_INCLUDE_DIRS})
        target_compile_options(opencv_test_liveview PRIVATE ${FFMPEG_CFLAGS_OTHER})
        target_link_libraries(opencv_test_liveview PRIVATE ${FFMPEG_LIBRARIES})
    endif()
    target_compile_definitions(opencv_test_liveview PRIVATE ${LIVEVIEW_DEFINES})
endif()

set(_yn_civet "NO")
set(_yn_boost "NO")
if(CIVETWEB_FOUND)
    set(_yn_civet "YES")
endif()
if(BOOST_FOUND)
    set(_yn_boost "YES")
endif()
set(_yn_ffmpeg "NO")
set(_yn_video "NO")
if(FFMPEG_FOUND)
    set(_yn_ffmpeg "YES")
endif()
if(LIVEVIEW_VIDEO_ENCODER_ENABLED)
    set(_yn_video "YES")
endif()
message(STATUS "\nLiveView configuration summary:\n  HTTP backend preference  : ${LIVEVIEW_HTTP_BACKEND}\n  HTTP backend selected    : ${LIVEVIEW_SELECTED_HTTP_BACKEND}\n  CivetWeb found           : ${_yn_civet}\n  Boost (Beast) found      : ${_yn_boost}\n  Video encoder preference : ${LIVEVIEW_VIDEO_ENCODER}\n  FFmpeg found             : ${_yn_ffmpeg}\n  Video encoder enabled    : ${_yn_video}\n")
if(DEFINED _liveview_backend_warn)
    message(WARNING "${_liveview_backend_warn}")
endif()
